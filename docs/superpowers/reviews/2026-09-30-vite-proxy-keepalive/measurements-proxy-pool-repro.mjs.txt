import http from 'node:http';
import { createServer } from 'vite';
const seen = new Set();
const target = http.createServer((req, res) => { seen.add(req.socket.remotePort); res.setHeader('content-type', 'application/json'); res.end('{"ok":true}'); });
target.keepAliveTimeout = 65000;
await new Promise((r) => target.listen(0, '127.0.0.1', r));
const tport = target.address().port;
for (const withAgent of [false, true]) {
  seen.clear();
  const proxyOpts = { target: `http://127.0.0.1:${tport}`, ...(withAgent ? { agent: new http.Agent({ keepAlive: true, timeout: 30000 }) } : {}) };
  const vite = await createServer({ configFile: false, logLevel: 'silent', server: { port: 0, host: '127.0.0.1', proxy: { '/api': proxyOpts } } });
  await vite.listen();
  const vport = vite.httpServer.address().port;
  const client = new http.Agent({ keepAlive: true });
  let clientClose = 0;
  for (let i = 0; i < 20; i++) {
    await new Promise((resolve) => http.get({ host: '127.0.0.1', port: vport, path: '/api/x', agent: client }, (res) => { if (res.headers.connection === 'close') clientClose++; res.resume(); res.on('end', resolve); }));
  }
  console.log(`agent=${withAgent}: 20 requests -> ${seen.size} distinct proxy->app connections; responses to client with Connection: close = ${clientClose}`);
  client.destroy(); await vite.close();
}
target.close();
